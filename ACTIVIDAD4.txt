;MULTIPLICAR 2 NUMEROS
.Const

.Data?
;SE DECLARA VARIABLE PARA RESERVARLA Y CONEVERTIRLA EN STRING
	resultadotex Byte 5 Dup(?)
.Data
;SE DECLARA VARAIBLE CON VALOR 10 EN DECIMIMAL
;SE DECLARA VARIABLE PARA ALMACENAR EL RESULTADO
;SE DECLARA VARIABLE PARA MOSTRAR EL TITULO DE LA CAJA
	N1 DWord 0AH; decimal 10
	resultado DWord 0H
	titulo Byte "Multiplicacion", 0
.Code
start:
;SE AGREGA EL VALOR 2 EN ESPACIO DE LA MEMORIA DEL EQUIPO
;SE REALIZA LA MULTIPLICACION DEL VALOR DE LA VARIABLE N1
;SE ALMACENA EL VALOR DE LA OPERCION EN LA VARIABLE RESULTADO
	Mov Eax, 2
	IMul Eax, N1
	Mov resultado, Eax
;SE REALIZA LA CONVERSION DE DATO TIPO ENTERO A TIPO STRING
	Invoke dwtoa, resultado, Addr resultadotex
;SE MUESTRA EL MENSAJE DEL RESULTADO FINAL
	Invoke MessageBox, 0, Addr resultadotex, Addr titulo, MB_OK
;LE INDICAMOS AL PROGRMAMA QUE AQUI TIENE QUE TERMINAR
	Invoke ExitProcess, 0
End start
