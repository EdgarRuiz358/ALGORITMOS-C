;SUMA DE DOS NUMEROS

.Const
.Data?
	;DECLARAMOS LA VARIABLE DONE NO SE INICIALIZA DATO PERO SI QUEDA RESERVADA EN LA MEMORIA
	resultadoentexto Byte 5 Dup(?)
.Data
	;DECLARAMOS LA VARIABLE NUMERO 1 CON DATO EN DECIMAL 10
	;DECLARAMOS LA VARIABLE NUMERO 2 CON DATO EN DECIMAL 15
	;DECLARAMOS LA VARIABLE RESULTADO AQUI ES DONDE SE ALMACENA LA SUMA DE AMBOS NUMEROS
	;SE LE AGREGA LA VARIABLE PARA EL NOMBRE DEL ENCABEZADO DE LA CAJA DE MENSAJE
	numerouno DWord 0AH
	numerodos DWord 0FH
	resultado DWord 0H
	Titulo Byte "EL RESULTADO DE LA SUMA ES", 0
.Code
start:
	;INICIALMENTE MOVEMOS EN UN ESPACIO DE LA MEMORIA LA VARIABLE NUMEROUNO YA QUE NO SE PUEDEN SUMAR COMO SE CONOCEN EN LOS LENGUAJES DE PROGRAMACION DE MAYOR NIVEL
	;UNA VEZ QUE SE MOVIO LA VARIABLE NUUMEROUNO PODEMOS HACER LA ADICION DEL NUMERO DOS CON ADD, AHI LE INDICAMOS AL PROGRAMA QUE LO SUME
	;CON MOV RESULTADO EL TOTAL DE LA SUMA SE ALAMCENA EN ESA VARIABLE
	Mov Eax, numerouno
	Add Eax, numerodos
	Mov resultado, Eax
	;EL DATO QUE DESEAMOS VER EL PROGRAMA POR SI SOLO NO LO INTERPRETA, SE TIENE QUE HACER UNA CONVERSION DE TIPO DE DATO ENTERO A STRING
	;PARA PODER VERLO EN LA CAJA DE MENSAJE, ESO SI TENEMOS QUE MANDAR A LLAMAR LA VARIABLE QUE ESTABA RESERVADA EN MEMORIA
	Invoke dwtoa, resultado, Addr resultadoentexto
	;AQUI NOS MOSTRARA EL RESULTADO EN DECIMAL
	Invoke MessageBox, 0, Addr resultadoentexto, Addr Titulo, MB_OK
	;FIN DE LA CAJA DE MENSAJE
	Invoke ExitProcess, 0
End start
