;RESTAR 3 A 10 CON SUB
.Const

.Data?
;INGRESAMO SLA VARIABLE QUE ALAMCENA EL VALOR A STRING
	resultadotex Byte 5 Dup(?)
.Data
;DECLARARMOS LAS VARIABLES
	N0 DWord 03H ;NUMERO  3 EN DECIMAL
	N1 DWord 0AH ;NUMERO 10 EN DECIMAL
	resultado DWord 0H;NUMERO 0 EN DECIMAL
	titulo Byte "Resta de dos numeros", 0;MOSTRAR EL TITULO DE MENSAJE

.Code

start:
;MOVEMOS LA VARIABLE N1 A ESPACIO DE MEMORIA
;REALIZAMOS LA RESTA 
;MOVEMOS LA VARIABLE RESULTADO PARA QUE ALAMCENE EL VALOR DE LA RESTA
	Mov Eax, N1
	Sub Eax, N0
	Mov resultado, Eax
;CONVERTIMOS EL VALOR ENTERO A TIPO STRING 
;REALIZAMOS LA MUSTRA DEL MENSAJE FINAL
	Invoke dwtoa, resultado, Addr resultadotex
	Invoke MessageBox, 0, Addr resultadotex, Addr titulo, MB_OK
;LE INDICAMOS AL PROGAMA QUE AQUI TERMINA
	Invoke ExitProcess, 0

End start
